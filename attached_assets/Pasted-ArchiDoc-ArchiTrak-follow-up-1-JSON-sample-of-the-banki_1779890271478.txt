const SENSITIVE_OBJECT_PREFIXES = [
  'contractors/',           // RIB documents (Task #491 banking)
  'insurance-certificates/', // Décennale / RC Pro certificates
];
app.get(/^\/objects\/(.+)$/, async (req, res, next) => {
  const entityId = req.params[0];
  const isSensitive = SENSITIVE_OBJECT_PREFIXES.some(p => entityId.startsWith(p));
  if (isSensitive) {
    return requireSessionOrSyncKey(req, res, next);
  }
  next();
}, async (req, res) => {

c. requireSessionOrSyncKey (server/middleware/requireSessionOrSyncKey.ts) — the single shared bearer accepted on the RIB proxy:

export const requireSessionOrSyncKey: RequestHandler = (req, res, next) => {
  // Path A — existing in-browser ArchiDoc users.
  if (req.session?.userId) return next();
  // Path B — sister-app server-to-server fetches.
  const authHeader = req.headers.authorization;
  if (authHeader && authHeader.startsWith('Bearer ')) {
    const apiKey = process.env.ARCHIDOC_SYNC_API_KEY;
    if (!apiKey) {
      console.error('[Auth] ARCHIDOC_SYNC_API_KEY is not configured — rejecting sync-key auth attempt');
      return res.status(503).json({ error: 'Sync API not configured' });
    }
    if (safeCompareSecret(authHeader.slice(7), apiKey)) return next();
  }
  return res.status(401).json({ error: 'Unauthorized' });
};
// safeCompareSecret: length guard + crypto.timingSafeEqual

requireSyncApiKey (used by the four /api/sync/* routes) is the same safeCompareSecret but bearer-only — no session path.

On the 404 contract you flagged: the object-proxy returns 404 JSON: { error: "Object not found" } rather than HTML when the GCS object is missing and the Replit fallback bucket also has no copy. Your "clean log on expired link" expectation holds.
